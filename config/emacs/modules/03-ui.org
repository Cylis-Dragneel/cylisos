** Dashboard
#+begin_src emacs-lisp
(use-package dashboard
  :ensure t
  :init
  (setq initial-buffer-choice 'dashboard-open
        dashboard-set-heading-icons t
        dashboard-set-file-icons t
        dashboard-banner-logo-title "Emacs Is A Goddamn Operating System!"
        dashboard-startup-banner (expand-file-name "images/emacs-dash.png" user-emacs-directory)
        dashboard-center-content t
        dashboard-projects-backend 'projectile
        dashboard-projects-switch-function 'projectile-switch-project-by-name
        dashboard-week-agenda t
        dashboard-agenda-prefix-format "%?-12t% s"
        dashboard-items '((agenda . 7)
                          (projects . 5)
                          (bookmarks . 5)
                          (registers . 5)
                          (recents . 5)))
  :custom
  (dashboard-modify-heading-icons '((recents . "file-text")
                                    (bookmarks . "book")))
  :config
  (dashboard-setup-startup-hook))
#+end_src

** Fonts

(set-face-attribute 'default nil
                    :font "Maple Mono NF CN Medium"
                    :height 110
                    :weight 'medium)

                    :font "Montserrat"
                    :height 120
                    :weight 'medium)

                    :font "Maple Mono NF CN Medium"
                    :height 110
                    :weight 'medium)

(set-fontset-font t 'unicode (font-spec :family "Symbola") nil 'append)
(setq use-default-font-for-symbols nil)
(when (member "Noto Color Emoji" (font-family-list))
  (set-fontset-font t 'symbol (font-spec :family "Noto Color Emoji") nil 'prepend))

(set-face-attribute 'font-lock-comment-face t :slant 'italic)
(set-face-attribute 'font-lock-keyword-face t :slant 'italic)
(setq org-src-fontify-natively t)

(add-to-list 'default-frame-alist '(font . "Maple Mono NF CN Medium-11"))
(setq-default line-spacing 0.12)

** Neotree
#+begin_src emacs-lisp
(use-package neotree
  :config
  (setq neo-smart-open t
        neo-show-hidden-files t
        neo-window-width 55
        neo-window-fixed-size nil
        inhibit-compacting-font-caches t
        projectile-switch-project-action 'neotree-projectile-action)
  (add-hook 'neo-after-create-hook
            #'(lambda (_)
                (with-current-buffer (get-buffer neo-buffer-name)
                  (setq truncate-lines t
                        word-wrap nil)
                  (make-local-variable 'auto-hscroll-mode)
                  (setq auto-hscroll-mode nil)))))
#+end_src
